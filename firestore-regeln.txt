rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /teams/{team}/members/{uid} {
      allow read: if request.auth != null;
      allow create, update: if request.auth != null && request.auth.uid == uid
                            && request.resource.data.j is string && request.resource.data.j.size() < 20000;
      allow delete: if request.auth != null && request.auth.uid == uid;
    }
    match /teams/{team}/kudos/{id} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
                    && request.resource.data.j is string && request.resource.data.j.size() < 1000;
    }
    match /teams/{team}/meta/{doc} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
                   && request.resource.data.j is string && request.resource.data.j.size() < 5000;
    }
  }
}
